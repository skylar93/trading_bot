```python
"""
Unified RiskManager that merges basic risk checks (drawdown, position size, daily trades)
with advanced portfolio risk (VaR, correlation, CVaR) from the older risk/risk_manager.py.
Positions are assumed to be Dict[str, Dict[str, float]]: 
  positions[symbol] = {"units": float, "avg_price": float, "cost_basis": float}
"""

import logging
import numpy as np
import pandas as pd
from dataclasses import dataclass
from typing import Dict, Optional, Any, Union


@dataclass
class RiskConfig:
    """Merged configuration for risk management."""
    # Basic risk
    max_position_size: float = 0.2       # fraction of portfolio
    stop_loss_pct: float = 0.02
    max_drawdown_pct: float = 0.15
    daily_trade_limit: int = 10
    min_trade_size: float = 0.01        # fraction of portfolio
    max_leverage: float = 1.0
    
    # Advanced portfolio risk
    volatility_lookback: int = 20
    risk_free_rate: float = 0.02
    var_confidence_level: float = 0.95
    correlation_window: int = 30
    max_correlation: float = 0.7
    portfolio_var_limit: float = 0.02   # fraction of portfolio value

class RiskManager:
    """
    Unified Risk Manager that handles:
      - daily trade limits
      - max drawdown
      - max leverage
      - position size clamp
      - min trade size
      - VaR / CVaR
      - correlation checks
    """
    def __init__(self, config: Optional[RiskConfig] = None):
        self.config = config or RiskConfig()
        self.logger = logging.getLogger(self.__class__.__name__)
        self.reset()

    def reset(self):
        """Reset all internal state."""
        # For daily trade limits
        self.trade_counter = {}  # {date: int}
        
        # For drawdown tracking
        self.peak_value: Optional[float] = None
        self.current_drawdown: float = 0.0
        
        # For advanced portfolio risk
        self._asset_returns: Dict[str, pd.Series] = {}  # price/returns data
        self._correlation_matrix: Optional[pd.DataFrame] = None
        self._last_correlation_update: Optional[pd.Timestamp] = None

    def check_trade(
        self,
        timestamp: pd.Timestamp,
        portfolio_value: float,
        trade_size: float,
        price: float,
        positions: Dict[str, Dict[str, float]],
        asset: str,
        price_data: Optional[Dict[str, pd.Series]] = None,
    ) -> Dict[str, Any]:
        """
        Main entry point: check if the proposed trade is allowed 
        and how big it can be (adjusted_size).

        Args:
            timestamp: current timestamp
            portfolio_value: current total portfolio value
            trade_size: proposed trade size (units)
            price: current price of this asset
            positions: dictionary of {symbol: {"units":..., "avg_price":..., "cost_basis":...}}
            asset: symbol being traded
            price_data: optional dict of full price series for advanced correlation/returns

        Returns:
            {
              "allowed": bool,
              "adjusted_size": float,
              "reason": str
            }
        """
        date_key = timestamp.date()
        
        # 1) Daily trade limit
        if date_key not in self.trade_counter:
            self.trade_counter[date_key] = 0
        if self.trade_counter[date_key] >= self.config.daily_trade_limit:
            self.logger.warning(f"Daily trade limit reached for {date_key}")
            return {
                "allowed": False,
                "adjusted_size": 0.0,
                "reason": "Daily trade limit reached"
            }

        # 2) Drawdown check
        if self.peak_value is not None:
            current_drawdown = (self.peak_value - portfolio_value) / self.peak_value
            if current_drawdown > self.config.max_drawdown_pct:
                self.logger.warning(
                    f"Max drawdown exceeded: {current_drawdown:.2%} > {self.config.max_drawdown_pct:.2%}"
                )
                return {
                    "allowed": False,
                    "adjusted_size": 0.0,
                    "reason": f"Max drawdown exceeded: {current_drawdown:.2%}"
                }

        # 3) Current + proposed exposure => leverage check
        #    We sum all existing positions plus the proposed new trade
        current_exposure = 0.0
        for sym, pos_dict in positions.items():
            # We only know the "units", so the 'price' might differ per symbol 
            # if multi-asset. In real usage, pass a dict of prices or store them somewhere.
            # For single asset, sym==asset => same price
            # Here, let's assume we do the same price for all? 
            # Or skip advanced multi-asset for example code.
            if sym == asset:
                # same asset => units * current price
                current_exposure += abs(pos_dict["units"] * price)
            else:
                # if truly multi-asset, you'd have to pass prices[sym]
                # for example code, just skip or treat as 0
                pass
        
        proposed_value = abs(trade_size * price)
        total_exposure = current_exposure + proposed_value
        if portfolio_value > 1e-8:  # avoid division by zero
            leverage = total_exposure / portfolio_value
            if leverage > self.config.max_leverage:
                # clamp trade_size
                max_allowed_exposure = self.config.max_leverage * portfolio_value
                remain_exposure = max_allowed_exposure - current_exposure
                if remain_exposure < 0:
                    # can't trade at all
                    return {
                        "allowed": False,
                        "adjusted_size": 0.0,
                        "reason": "Leverage limit reached"
                    }
                adjusted_size = remain_exposure / price
                if adjusted_size * price < self.config.min_trade_size * portfolio_value:
                    return {
                        "allowed": False,
                        "adjusted_size": 0.0,
                        "reason": "Leverage limit reached (clamped below min trade size)"
                    }
                return {
                    "allowed": True,
                    "adjusted_size": adjusted_size,
                    "reason": "Adjusted for leverage limit"
                }

        # 4) Position size clamp
        #    If the single trade_value is bigger than (max_position_size * portfolio)
        #    => clamp
        max_pos_value = self.config.max_position_size * portfolio_value
        if proposed_value > max_pos_value:
            adjusted_size = max_pos_value / price
            # check if after clamp it's still above the min
            if adjusted_size * price < self.config.min_trade_size * portfolio_value:
                return {
                    "allowed": False,
                    "adjusted_size": 0.0,
                    "reason": "Position size too large (below min size after clamp)"
                }
            return {
                "allowed": True,
                "adjusted_size": adjusted_size,
                "reason": "Clamped to max_position_size"
            }

        # 5) Minimum trade size
        min_value = self.config.min_trade_size * portfolio_value
        if proposed_value < min_value:
            return {
                "allowed": False,
                "adjusted_size": 0.0,
                "reason": "Trade size below minimum"
            }

        # 6) Optional advanced checks: e.g. portfolio VaR limit or correlation
        if price_data is not None:
            # If user passed in price_data => we can update correlation, etc.
            self.update_correlation_matrix(price_data)
            # check correlation with existing positions if needed
            # (omitted: see check_correlation_limits() usage example)

            # also check portfolio VaR
            # let's do a quick get_portfolio_var with the new position included
            hypothetical_positions = {}
            for sym, pos_dict in positions.items():
                # convert to float for advanced calc
                hypothetical_positions[sym] = pos_dict["units"]
            # add the new trade
            # if trade_size is positive => add; if negative => reduce
            if asset not in hypothetical_positions:
                hypothetical_positions[asset] = 0.0
            hypothetical_positions[asset] += trade_size

            # Then compute portfolio VaR
            portfolio_var = self.get_portfolio_var(
                hypothetical_positions, 
                portfolio_value
            )
            if portfolio_var > self.config.portfolio_var_limit:
                return {
                    "allowed": False,
                    "adjusted_size": 0.0,
                    "reason": f"Portfolio VaR limit exceeded: {portfolio_var:.2f}"
                }

        # Passed all checks
        return {
            "allowed": True,
            "adjusted_size": trade_size,
            "reason": "OK"
        }

    def update_after_trade(self, timestamp: pd.Timestamp) -> None:
        """Call after a trade is executed successfully to update daily count, etc."""
        date_key = timestamp.date()
        if date_key not in self.trade_counter:
            self.trade_counter[date_key] = 0
        self.trade_counter[date_key] += 1
        self.logger.debug(f"Trade counter for {date_key}: {self.trade_counter[date_key]}")

    def update_drawdown(self, portfolio_value: float) -> float:
        """Update internal peak_value and current_drawdown, return current drawdown."""
        if self.peak_value is None:
            self.peak_value = portfolio_value
            self.current_drawdown = 0.0
        elif portfolio_value > self.peak_value:
            self.peak_value = portfolio_value
            self.current_drawdown = 0.0
        else:
            self.current_drawdown = (self.peak_value - portfolio_value) / self.peak_value

            if self.current_drawdown > self.config.max_drawdown_pct:
                self.logger.warning(
                    f"Max drawdown exceeded: {self.current_drawdown:.2%} > {self.config.max_drawdown_pct:.2%}"
                )
        return self.current_drawdown

    # =====================
    # ADVANCED METHODS
    # =====================

    def calculate_var(self, returns: pd.Series, confidence_level: float) -> float:
        """
        Calculate Value at Risk from historical returns at given confidence level.
        returns: series of daily (or period) returns
        confidence_level: e.g. 0.95
        """
        if len(returns) < 2:
            return 0.0
        # simplistic: percentile based VaR
        return abs(np.percentile(returns, (1 - confidence_level) * 100))

    def calculate_cvar(self, returns: pd.Series, confidence_level: float) -> float:
        """
        Calculate Conditional Value at Risk (CVaR).
        This is the average of the losses beyond the VaR threshold.
        """
        var = self.calculate_var(returns, confidence_level)
        if len(returns) < 2 or var <= 1e-12:
            return 0.0
        # negative returns <= -var
        tail = returns[returns <= -var]
        if len(tail) == 0:
            return 0.0
        return abs(tail.mean())

    def update_correlation_matrix(self, price_data: Dict[str, pd.Series]) -> None:
        """
        Recompute correlation matrix from the provided price_data.
        price_data: {asset: pd.Series of prices} 
                    We'll compute returns over 'correlation_window' lookback
        """
        if not price_data:
            return
        
        # Build returns
        for asset, prices in price_data.items():
            # store or update self._asset_returns
            self._asset_returns[asset] = prices.pct_change().dropna()

        # Combine into dataframe w/ same window
        returns_df = {}
        for asset, rets in self._asset_returns.items():
            windowed = rets.tail(self.config.correlation_window)
            if len(windowed) > 1:
                returns_df[asset] = windowed
        if not returns_df:
            return
        
        df = pd.DataFrame(returns_df)
        if df.shape[1] < 2:
            return  # only 1 asset => no correlation to compute

        self._correlation_matrix = df.corr()
        self._last_correlation_update = pd.Timestamp.now()

    def check_correlation_limits(self, asset1: str, asset2: str) -> bool:
        """
        True if correlation between asset1 & asset2 is <= max_correlation
        """
        if self._correlation_matrix is None:
            return True  # no data => can't check => allow

        if (asset1 not in self._correlation_matrix.columns 
            or asset2 not in self._correlation_matrix.columns):
            return True  # no data => allow

        corr = abs(self._correlation_matrix.loc[asset1, asset2])
        return (corr <= self.config.max_correlation)

    def get_portfolio_var(
        self, 
        positions: Dict[str, float],
        portfolio_value: float
    ) -> float:
        """
        Approximate portfolio VaR by combining individual asset VaRs 
        with correlation (if available). 
        positions: {asset: float(units)} - for advanced calc, we need to approximate the asset's $ value or returns 
        portfolio_value: total portfolio value 
        """
        if not positions or portfolio_value <= 0:
            return 0.0

        # 1) Compute weights
        #    We do NOT know each asset's price from just `positions`.
        #    This example code uses the last daily return in self._asset_returns to approximate "latest_price"
        #    => you'll need to adapt for real usage.
        weights = {}
        for asset, units in positions.items():
            # We guess the "latest price" as 1 if we have no better info
            # Or you'd store a map {asset: current_price} somewhere
            # For demonstration, let's just do "units" => skip
            # Real usage: you'd pass a dict of {asset: current_price} => 
            #   position_value = units * current_price
            #   weight = position_value / portfolio_value
            # For now, do something naive:
            position_value = abs(units) * 1.0  # naive
            weights[asset] = position_value / portfolio_value

        # 2) gather each asset's historical returns & compute VaR
        asset_vars = []
        assets_in_matrix = []
        for asset, w in weights.items():
            if asset in self._asset_returns and w > 0:
                var = self.calculate_var(
                    self._asset_returns[asset],
                    self.config.var_confidence_level
                )
                asset_vars.append(var)
                assets_in_matrix.append(asset)
            else:
                asset_vars.append(0.0)
                assets_in_matrix.append(asset)

        # if all zero => no var
        if all(v == 0.0 for v in asset_vars):
            return 0.0

        # 3) build var-cov matrix
        var_diag = np.diag(asset_vars)
        weight_array = np.array([weights[a] for a in assets_in_matrix])
        
        # 4) incorporate correlation
        if self._correlation_matrix is not None and len(assets_in_matrix) > 1:
            subset_corr = self._correlation_matrix.reindex(
                index=assets_in_matrix, 
                columns=assets_in_matrix
            ).fillna(0.0).values
            # combine
            # a very naive way to incorporate correlation with var:
            portfolio_var = np.sqrt(weight_array.dot(var_diag).dot(subset_corr).dot(weight_array.T))
        else:
            # no correlation => sum of weighted VaRs
            portfolio_var = np.sqrt(weight_array.dot(var_diag).dot(weight_array.T))

        # This is just a relative measure. Cap at 1.0 => 100%
        return min(float(portfolio_var), 1.0)

    # etc... (other advanced methods if needed)
```

### 코드 사용 요령

1. **백테스터(`BaseBacktester`) 쪽에서** `risk_manager = RiskManager(risk_config)` 식으로 생성.  
2. **`execute_trade`나 `update` 메서드**에서 거래 직전:
   ```python
   risk_check = self.risk_manager.check_trade(
       timestamp=timestamp,
       portfolio_value=current_portfolio_value,
       trade_size=trade_amount,
       price=current_price,
       positions=self.positions,  # dict[str, {units, avg_price, cost_basis}]
       asset=symbol,
       price_data=...  # (선택) 자산별 pd.Series 있으면 넣기
   )
   if not risk_check["allowed"]:
       # 거절 처리
       trade["success"] = False
       trade["reason"] = risk_check["reason"]
       return trade
   
   # clamp
   if abs(risk_check["adjusted_size"] - trade_amount) > 1e-8:
       trade_amount = risk_check["adjusted_size"]
       ...
   ```
3. 체결 후 `self.risk_manager.update_after_trade(timestamp)` 를 호출하여, 일일 거래 횟수 증가.  
4. 매 스텝에서 `portfolio_value`가 갱신될 때 `self.risk_manager.update_drawdown(portfolio_value)` 로 드로다운 갱신.  

이렇게 하면:
- 기존 `max_drawdown_pct`, `daily_trade_limit`, `max_leverage`, `max_position_size`, `min_trade_size` 등 **기본 리스크**를 모두 체크  
- `VaR/상관관계` 같은 **고급 기능**도 `price_data`를 넘겨주면 `update_correlation_matrix()`가 자동 업데이트, `check_trade`에서 `get_portfolio_var()`로 포트폴리오 VaR 초과 여부를 검사 가능합니다.

> 실제 운영 시, 여러 자산에 대해 시세 `price_data`(예: `{ "BTC": pd.Series([...]) , "ETH": pd.Series([...]) }`)를 주기적으로 업데이트하고, `positions[sym]["units"] * current_price[sym]` 식으로 각 자산의 시장가치를 구해야 “정확한 레버리지/노출”을 계산할 수 있습니다. 여기서는 **예시**로만 보여드렸습니다.

```
      - daily trade limits
      - max drawdown
      - max leverage

      - correlation checks

        self._correlation_matrix: Optional[pd.DataFrame] = None
        self._last_correlation_update: Optional[pd.Timestamp] = None

    def check_trade(
        self,
        timestamp: pd.Timestamp,
        portfolio_value: float,
        trade_size: float,
        price: float,
        positions: Dict[str, Dict[str, float]],
        asset: str,
        price_data: Optional[Dict[str, pd.Series]] = None,
    ) -> Dict[str, Any]:
        """
        Main entry point: check if the proposed trade is allowed 
        and how big it can be (adjusted_size).

        Args:
            timestamp: current timestamp
            portfolio_value: current total portfolio value
            trade_size: proposed trade size (units)
            price: current price of this asset
            positions: dictionary of {symbol: {"units":..., "avg_price":..., "cost_basis":...}}
            asset: symbol being traded
            price_data: optional dict of full price series for advanced correlation/returns

        Returns:
            {
              "allowed": bool,
              "adjusted_size": float,
              "reason": str
            }
        """

        # 1) Daily trade limit
        if date_key not in self.trade_counter:
            self.trade_counter[date_key] = 0
        if self.trade_counter[date_key] >= self.config.daily_trade_limit:
            self.logger.warning(f"Daily trade limit reached for {date_key}")
            return {
                "allowed": False,
                "adjusted_size": 0.0,
                "reason": "Daily trade limit reached"
            }

        # 2) Drawdown check
        if self.peak_value is not None:
            current_drawdown = (self.peak_value - portfolio_value) / self.peak_value
            if current_drawdown > self.config.max_drawdown_pct:
                self.logger.warning(
                    f"Max drawdown exceeded: {current_drawdown:.2%} > {self.config.max_drawdown_pct:.2%}"
                )
                return {
                    "allowed": False,
                    "adjusted_size": 0.0,
                    "reason": f"Max drawdown exceeded: {current_drawdown:.2%}"

            # if multi-asset. In real usage, pass a dict of prices or store them somewhere.
            # For single asset, sym==asset => same price
            if sym == asset:
                # same asset => units * current price
                current_exposure += abs(pos_dict["units"] * price)
            else:
                # if truly multi-asset, you'd have to pass prices[sym]
                # for example code, just skip or treat as 0
                pass
        
        proposed_value = abs(trade_size * price)
        total_exposure = current_exposure + proposed_value
        if portfolio_value > 1e-8:  # avoid division by zero
            leverage = total_exposure / portfolio_value
            if leverage > self.config.max_leverage:
                # clamp trade_size
                max_allowed_exposure = self.config.max_leverage * portfolio_value
                remain_exposure = max_allowed_exposure - current_exposure
                if remain_exposure < 0:
                    # can't trade at all
                    return {
                        "allowed": False,
                        "adjusted_size": 0.0,
                        "reason": "Leverage limit reached"
                    }
                adjusted_size = remain_exposure / price
                if adjusted_size * price < self.config.min_trade_size * portfolio_value:
                    return {
                        "allowed": False,
                        "adjusted_size": 0.0,
                        "reason": "Leverage limit reached (clamped below min trade size)"
                    }
                return {
                    "allowed": True,
                    "adjusted_size": adjusted_size,
                    "reason": "Adjusted for leverage limit"
                }

        # 4) Position size clamp
        #    If the single trade_value is bigger than (max_position_size * portfolio)
        #    => clamp
        max_pos_value = self.config.max_position_size * portfolio_value
        if proposed_value > max_pos_value:
            adjusted_size = max_pos_value / price
            # check if after clamp it's still above the min
            if adjusted_size * price < self.config.min_trade_size * portfolio_value:
                return {
                    "allowed": False,
                    "adjusted_size": 0.0,
                    "reason": "Position size too large (below min size after clamp)"
                }
            return {
                "allowed": True,
                "adjusted_size": adjusted_size,
                "reason": "Clamped to max_position_size"
            }

        # 5) Minimum trade size
        min_value = self.config.min_trade_size * portfolio_value
        if proposed_value < min_value:
            return {
                "allowed": False,
                "adjusted_size": 0.0,
                "reason": "Trade size below minimum"
            }

        # 6) Optional advanced checks: e.g. portfolio VaR limit or correlation
        if price_data is not None:
            # If user passed in price_data => we can update correlation, etc.
            self.update_correlation_matrix(price_data)
            # check correlation with existing positions if needed
            # (omitted: see check_correlation_limits() usage example)

            # also check portfolio VaR
            # let's do a quick get_portfolio_var with the new position included
            hypothetical_positions = {}
            for sym, pos_dict in positions.items():
                # convert to float for advanced calc
                hypothetical_positions[sym] = pos_dict["units"]
            # add the new trade
            # if trade_size is positive => add; if negative => reduce
            if asset not in hypothetical_positions:
                hypothetical_positions[asset] = 0.0
            hypothetical_positions[asset] += trade_size

            # Then compute portfolio VaR
            portfolio_var = self.get_portfolio_var(
                hypothetical_positions, 
                portfolio_value
            )
            if portfolio_var > self.config.portfolio_var_limit:
                return {
                    "allowed": False,
                    "adjusted_size": 0.0,
                    "reason": f"Portfolio VaR limit exceeded: {portfolio_var:.2f}"
                }

        # Passed all checks
        return {
            "allowed": True,
            "adjusted_size": trade_size,
            "reason": "OK"
        }

    def update_after_trade(self, timestamp: pd.Timestamp) -> None:
        """Call after a trade is executed successfully to update daily count, etc."""
        date_key = timestamp.date()
        if date_key not in self.trade_counter:
            self.trade_counter[date_key] = 0
        self.trade_counter[date_key] += 1
        self.logger.debug(f"Trade counter for {date_key}: {self.trade_counter[date_key]}")

    def update_drawdown(self, portfolio_value: float) -> float:
        """Update internal peak_value and current_drawdown, return current drawdown."""
        if self.peak_value is None:
            self.peak_value = portfolio_value
            self.current_drawdown = 0.0
        elif portfolio_value > self.peak_value:
            self.peak_value = portfolio_value
            self.current_drawdown = 0.0
        else:
            self.current_drawdown = (self.peak_value - portfolio_value) / self.peak_value

            if self.current_drawdown > self.config.max_drawdown_pct:
                self.logger.warning(
                    f"Max drawdown exceeded: {self.current_drawdown:.2%} > {self.config.max_drawdown_pct:.2%}"
                )
        return self.current_drawdown

    # =====================
    # ADVANCED METHODS
    # =====================

    def calculate_var(self, returns: pd.Series, confidence_level: float) -> float:
        """
        Calculate Value at Risk from historical returns at given confidence level.
        returns: series of daily (or period) returns
        confidence_level: e.g. 0.95
        """
        if len(returns) < 2:
            return 0.0
        # simplistic: percentile based VaR
        return abs(np.percentile(returns, (1 - confidence_level) * 100))

    def calculate_cvar(self, returns: pd.Series, confidence_level: float) -> float:
        """
        Calculate Conditional Value at Risk (CVaR).
        This is the average of the losses beyond the VaR threshold.
        """
        var = self.calculate_var(returns, confidence_level)
        if len(returns) < 2 or var <= 1e-12:
            return 0.0
        # negative returns <= -var
        tail = returns[returns <= -var]
        if len(tail) == 0:
            return 0.0
        return abs(tail.mean())

    def update_correlation_matrix(self, price_data: Dict[str, pd.Series]) -> None:
        """
        Recompute correlation matrix from the provided price_data.
        price_data: {asset: pd.Series of prices} 
                    We'll compute returns over 'correlation_window' lookback
        """
        if not price_data:
            return
        
        # Build returns
        for asset, prices in price_data.items():
            # store or update self._asset_returns
            self._asset_returns[asset] = prices.pct_change().dropna()

        # Combine into dataframe w/ same window
        returns_df = {}
        for asset, rets in self._asset_returns.items():
            windowed = rets.tail(self.config.correlation_window)
            if len(windowed) > 1:
                returns_df[asset] = windowed
        if not returns_df:
            return
        
        df = pd.DataFrame(returns_df)
        if df.shape[1] < 2:
            return  # only 1 asset => no correlation to compute

        self._correlation_matrix = df.corr()
        self._last_correlation_update = pd.Timestamp.now()

    def check_correlation_limits(self, asset1: str, asset2: str) -> bool:
        """
        True if correlation between asset1 & asset2 is <= max_correlation
        """
        if self._correlation_matrix is None:
            return True  # no data => can't check => allow

        if (asset1 not in self._correlation_matrix.columns 
            or asset2 not in self._correlation_matrix.columns):
            return True  # no data => allow

        corr = abs(self._correlation_matrix.loc[asset1, asset2])
        return (corr <= self.config.max_correlation)

    def get_portfolio_var(
        self, 
        positions: Dict[str, float],
        portfolio_value: float
    ) -> float:
        """
        Approximate portfolio VaR by combining individual asset VaRs 
        with correlation (if available). 
        positions: {asset: float(units)} - for advanced calc, we need to approximate the asset's $ value or returns 
        portfolio_value: total portfolio value 
        """
        if not positions or portfolio_value <= 0:
            return 0.0

        # 1) Compute weights
        #    We do NOT know each asset's price from just `positions`.
        #    This example code uses the last daily return in self._asset_returns to approximate "latest_price"
        #    => you'll need to adapt for real usage.
        weights = {}
        for asset, units in positions.items():
            # We guess the "latest price" as 1 if we have no better info
            # Or you'd store a map {asset: current_price} somewhere
            # For demonstration, let's just do "units" => skip
            # Real usage: you'd pass a dict of {asset: current_price} => 
            #   position_value = units * current_price
            #   weight = position_value / portfolio_value
            # For now, do something naive:
            position_value = abs(units) * 1.0  # naive
            weights[asset] = position_value / portfolio_value

        # 2) gather each asset's historical returns & compute VaR
        asset_vars = []
        assets_in_matrix = []
        for asset, w in weights.items():
            if asset in self._asset_returns and w > 0:
                var = self.calculate_var(
                    self._asset_returns[asset],
                    self.config.var_confidence_level
                )
                asset_vars.append(var)
                assets_in_matrix.append(asset)
            else:
                asset_vars.append(0.0)
                assets_in_matrix.append(asset)

        # if all zero => no var
        if all(v == 0.0 for v in asset_vars):
            return 0.0

        # 3) build var-cov matrix
        var_diag = np.diag(asset_vars)
        weight_array = np.array([weights[a] for a in assets_in_matrix])
        
        # 4) incorporate correlation
        if self._correlation_matrix is not None and len(assets_in_matrix) > 1:
            subset_corr = self._correlation_matrix.reindex(
                index=assets_in_matrix, 
                columns=assets_in_matrix
            ).fillna(0.0).values
            # combine
            # a very naive way to incorporate correlation with var:
            portfolio_var = np.sqrt(weight_array.dot(var_diag).dot(subset_corr).dot(weight_array.T))
        else:
            # no correlation => sum of weighted VaRs
            portfolio_var = np.sqrt(weight_array.dot(var_diag).dot(weight_array.T))

        # This is just a relative measure. Cap at 1.0 => 100%
        return min(float(portfolio_var), 1.0)
```

```python
"""
Unified RiskManager that merges basic risk checks (drawdown, position size, daily trades)
with advanced portfolio risk (VaR, correlation, CVaR).
Positions are assumed to be Dict[str, Dict[str, float]]:
  positions[symbol] = {"units": float, "avg_price": float, "cost_basis": float}
Also includes legacy methods (check_max_drawdown, calculate_stop_loss, etc.)
to satisfy older tests in test_risk_management.py
"""

import logging
import numpy as np
import pandas as pd
from dataclasses import dataclass
from typing import Dict, Optional, Any, Union


@dataclass
class RiskConfig:
    """Merged configuration for risk management."""
    # Basic risk
    max_position_size: float = 0.2       # fraction of portfolio
    stop_loss_pct: float = 0.02
    max_drawdown_pct: float = 0.15
    daily_trade_limit: int = 10
    min_trade_size: float = 0.01        # fraction of portfolio
    max_leverage: float = 1.0
    
    # Advanced portfolio risk
    volatility_lookback: int = 20
    risk_free_rate: float = 0.02
    var_confidence_level: float = 0.95
    correlation_window: int = 30
    max_correlation: float = 0.7
    portfolio_var_limit: float = 0.02   # fraction of portfolio value


class RiskManager:
    """
    Unified Risk Manager that handles:
      - daily trade limits
      - max drawdown
      - max leverage
      - position size clamp
      - min trade size
      - VaR / CVaR
      - correlation checks
      - plus legacy methods (check_trade_limits, check_max_drawdown, etc.)
        to satisfy older tests
    """
    def __init__(self, config: Optional[RiskConfig] = None):
        self.config = config or RiskConfig()
        self.logger = logging.getLogger(self.__class__.__name__)
        self.reset()

    def reset(self):
        """Reset all internal state."""
        # For daily trade limits
        self.trade_counter = {}  # {date: int}
        
        # For drawdown tracking
        self.peak_value: Optional[float] = None
        self.current_drawdown: float = 0.0
        
        # For advanced portfolio risk
        self._asset_returns: Dict[str, pd.Series] = {}  # price/returns data
        self._correlation_matrix: Optional[pd.DataFrame] = None
        self._last_correlation_update: Optional[pd.Timestamp] = None

    # -----------------------------------------------------------------------
    #  Existing "check_trade" used by BaseBacktester
    # -----------------------------------------------------------------------
    def check_trade(
        self,
        timestamp: pd.Timestamp,
        portfolio_value: float,
        trade_size: float,
        price: float,
        positions: Dict[str, Dict[str, float]],
        asset: str,
        price_data: Optional[Dict[str, pd.Series]] = None,
    ) -> Dict[str, Any]:
        """
        Main entry point: check if the proposed trade is allowed
        and how big it can be (adjusted_size).
        """
        date_key = timestamp.date()
        
        # 1) Daily trade limit
        if date_key not in self.trade_counter:
            self.trade_counter[date_key] = 0
        if self.trade_counter[date_key] >= self.config.daily_trade_limit:
            self.logger.warning(f"Daily trade limit reached for {date_key}")
            return {
                "allowed": False,
                "adjusted_size": 0.0,
                "reason": "Daily trade limit reached"
            }

        # 2) Drawdown check
        if self.peak_value is not None:
            current_drawdown = (self.peak_value - portfolio_value) / self.peak_value
            if current_drawdown > self.config.max_drawdown_pct:
                self.logger.warning(
                    f"Max drawdown exceeded: {current_drawdown:.2%} > {self.config.max_drawdown_pct:.2%}"
                )
                return {
                    "allowed": False,
                    "adjusted_size": 0.0,
                    "reason": f"Max drawdown exceeded: {current_drawdown:.2%}"
                }

        # 3) Leverage check
        current_exposure = 0.0
        for sym, pos_dict in positions.items():
            if sym == asset:
                current_exposure += abs(pos_dict["units"] * price)
            else:
                # For multi-asset, need that asset's price. Omitted in example.
                pass
        
        proposed_value = abs(trade_size * price)
        total_exposure = current_exposure + proposed_value
        if portfolio_value > 1e-8:
            leverage = total_exposure / portfolio_value
            if leverage > self.config.max_leverage:
                max_allowed_exposure = self.config.max_leverage * portfolio_value
                remain_exposure = max_allowed_exposure - current_exposure
                if remain_exposure < 0:
                    return {
                        "allowed": False,
                        "adjusted_size": 0.0,
                        "reason": "Leverage limit reached"
                    }
                adjusted_size = remain_exposure / price
                if adjusted_size * price < self.config.min_trade_size * portfolio_value:
                    return {
                        "allowed": False,
                        "adjusted_size": 0.0,
                        "reason": "Leverage limit reached (clamped below min trade size)"
                    }
                return {
                    "allowed": True,
                    "adjusted_size": adjusted_size,
                    "reason": "Adjusted for leverage limit"
                }

        # 4) Position size clamp
        max_pos_value = self.config.max_position_size * portfolio_value
        if proposed_value > max_pos_value:
            adjusted_size = max_pos_value / price
            if adjusted_size * price < self.config.min_trade_size * portfolio_value:
                return {
                    "allowed": False,
                    "adjusted_size": 0.0,
                    "reason": "Position size too large (below min size after clamp)"
                }
            return {
                "allowed": True,
                "adjusted_size": adjusted_size,
                "reason": "Clamped to max_position_size"
            }

        # 5) Minimum trade size
        min_value = self.config.min_trade_size * portfolio_value
        if proposed_value < min_value:
            return {
                "allowed": False,
                "adjusted_size": 0.0,
                "reason": "Trade size below minimum"
            }

        # 6) Optional advanced checks: portfolio VaR, correlation, etc.
        if price_data is not None:
            self.update_correlation_matrix(price_data)
            
            # Check portfolio VaR with hypothetical new position
            hypothetical_positions = {}
            for sym, pos_dict in positions.items():
                hypothetical_positions[sym] = pos_dict["units"]
            if asset not in hypothetical_positions:
                hypothetical_positions[asset] = 0.0
            hypothetical_positions[asset] += trade_size

            portfolio_var = self.get_portfolio_var(hypothetical_positions, portfolio_value)
            if portfolio_var > self.config.portfolio_var_limit:
                return {
                    "allowed": False,
                    "adjusted_size": 0.0,
                    "reason": f"Portfolio VaR limit exceeded: {portfolio_var:.2f}"
                }

        # Passed all checks
        return {
            "allowed": True,
            "adjusted_size": trade_size,
            "reason": "OK"
        }

    def update_after_trade(self, timestamp: pd.Timestamp) -> None:
        """Call after a trade is executed successfully to update daily count, etc."""
        date_key = timestamp.date()
        if date_key not in self.trade_counter:
            self.trade_counter[date_key] = 0
        self.trade_counter[date_key] += 1
        self.logger.debug(f"Trade counter for {date_key}: {self.trade_counter[date_key]}")

    def update_drawdown(self, portfolio_value: float) -> float:
        """Update internal peak_value and current_drawdown, return current drawdown."""
        if self.peak_value is None:
            self.peak_value = portfolio_value
            self.current_drawdown = 0.0
        elif portfolio_value > self.peak_value:
            self.peak_value = portfolio_value
            self.current_drawdown = 0.0
        else:
            self.current_drawdown = (self.peak_value - portfolio_value) / self.peak_value
            if self.current_drawdown > self.config.max_drawdown_pct:
                self.logger.warning(
                    f"Max drawdown exceeded: {self.current_drawdown:.2%} > {self.config.max_drawdown_pct:.2%}"
                )
        return self.current_drawdown

    # -----------------------------------------------------------------------
    #  Legacy or Additional methods (for older tests)
    # -----------------------------------------------------------------------
    def check_trade_limits(self, timestamp: pd.Timestamp) -> bool:
        """
        Return True if we have not exceeded the daily trade limit on this date.
        test_trade_limits() calls this.
        """
        date_key = timestamp.date()
        if date_key not in self.trade_counter:
            self.trade_counter[date_key] = 0
        # if we've hit the limit, return False
        if self.trade_counter[date_key] >= self.config.daily_trade_limit:
            self.logger.warning(f"Daily trade limit reached for {date_key}")
            return False
        return True

    def check_max_drawdown(self, peak_value: float, current_value: float) -> bool:
        """
        Return True if drawdown is beyond config.max_drawdown_pct.
        test_drawdown_monitoring() calls this with (initial_value, current_value).
        """
        if peak_value <= 0:
            return False
        dd = (peak_value - current_value) / peak_value
        return dd > self.config.max_drawdown_pct

    def check_leverage_limits(self, portfolio_value: float, position_value: float) -> bool:
        """
        Return True if position_value/portfolio_value <= max_leverage.
        test_leverage_limits() calls this.
        """
        if portfolio_value <= 0:
            return False
        leverage = position_value / portfolio_value
        return leverage <= self.config.max_leverage

    def calculate_stop_loss(
        self, entry_price: float, position_size: float, is_long: bool = True
    ) -> float:
        """
        Return the stop loss price based on config.stop_loss_pct.
        test_stop_loss() calls this.
        """
        if is_long:
            return entry_price * (1 - self.config.stop_loss_pct)
        else:
            return entry_price * (1 + self.config.stop_loss_pct)

    def process_trade_signal(self, signal: dict) -> bool:
        """
        Simple old interface: if trade signal has {timestamp, type, price, size}
        and price>0, size>0 => return True. Otherwise False.
        test_trade_signal_processing() calls this.
        """
        required_fields = ["timestamp", "type", "price", "size"]
        if not all(field in signal for field in required_fields):
            self.logger.warning("Invalid trade signal: missing required fields.")
            return False
        if signal["price"] <= 0 or signal["size"] <= 0:
            self.logger.warning("Invalid trade signal: negative price or zero size.")
            return False
        return True

    def calculate_position_size(
        self,
        portfolio_value: float,
        price: float,
        volatility: Optional[float] = None,
        current_positions: Optional[Dict[str, float]] = None,
        asset_name: Optional[str] = None
    ) -> float:
        """
        Return a NOTIONAL position size in dollars (not units).
        test_position_sizing() expects "portfolio_value * max_position_size" as a baseline.
        Also scale down if volatility is high or if correlation is too high.
        """
        # Basic position size limit in nominal terms
        position_size = portfolio_value * self.config.max_position_size

        # Adjust for volatility if provided
        if volatility is not None and volatility > 0:
            vol_scalar = 1.0 / (1.0 + volatility)
            position_size *= vol_scalar

        # If we have correlation data and an asset_name & positions
        if current_positions and asset_name and self._correlation_matrix is not None:
            # If correlation is too high, reduce the size. (Simplistic approach)
            for pos_asset, pos_value in current_positions.items():
                if not self.check_correlation_limits(asset_name, pos_asset):
                    # E.g. reduce by half
                    position_size *= 0.5

        # Enforce minimum size (in notional)
        min_notional = portfolio_value * self.config.min_trade_size
        if position_size < min_notional:
            return 0.0

        return position_size

    # =====================
    # ADVANCED METHODS
    # =====================

    def calculate_var(self, returns: pd.Series, confidence_level: float) -> float:
        """
        Calculate Value at Risk from historical returns at given confidence level.
        """
        if len(returns) < 2:
            return 0.0
        return abs(np.percentile(returns, (1 - confidence_level) * 100))

    def calculate_cvar(self, returns: pd.Series, confidence_level: float) -> float:
        """
        Calculate Conditional Value at Risk (CVaR).
        """
        var = self.calculate_var(returns, confidence_level)
        if len(returns) < 2 or var <= 1e-12:
            return 0.0
        tail = returns[returns <= -var]
        if len(tail) == 0:
            return 0.0
        return abs(tail.mean())

    def update_correlation_matrix(self, price_data: Dict[str, pd.Series]) -> None:
        """
        Recompute correlation matrix from the provided price_data.
        """
        if not price_data:
            return
        
        # Build returns
        for asset, prices in price_data.items():
            self._asset_returns[asset] = prices.pct_change().dropna()

        # Combine into dataframe w/ same window
        returns_df = {}
        for asset, rets in self._asset_returns.items():
            windowed = rets.tail(self.config.correlation_window)
            if len(windowed) > 1:
                returns_df[asset] = windowed
        if not returns_df:
            return
        
        df = pd.DataFrame(returns_df)
        if df.shape[1] < 2:
            return  # only 1 asset => no correlation to compute

        self._correlation_matrix = df.corr()
        self._last_correlation_update = pd.Timestamp.now()

    def check_correlation_limits(self, asset1: str, asset2: str) -> bool:
        """
        True if correlation between asset1 & asset2 is <= max_correlation
        Test expects a bool type (not numpy.bool_).
        """
        if self._correlation_matrix is None:
            return True
        if (asset1 not in self._correlation_matrix.columns
            or asset2 not in self._correlation_matrix.columns):
            return True

        corr = abs(self._correlation_matrix.loc[asset1, asset2])
        return bool(corr <= self.config.max_correlation)

    def get_portfolio_var(
        self,
        positions: Dict[str, float],
        portfolio_value: float
    ) -> float:
        """
        Approximate portfolio VaR by combining individual asset VaRs
        with correlation (if available).
        """
        if not positions or portfolio_value <= 0:
            return 0.0

        # 1) Compute weights (naive approach)
        weights = {}
        for asset, units in positions.items():
            position_value = abs(units) * 1.0
            weights[asset] = position_value / portfolio_value

        # 2) gather each asset's returns -> compute VaR
        asset_vars = []
        assets_in_matrix = []
        for asset, w in weights.items():
            if asset in self._asset_returns and w > 0:
                var = self.calculate_var(
                    self._asset_returns[asset],
                    self.config.var_confidence_level
                )
                asset_vars.append(var)
                assets_in_matrix.append(asset)
            else:
                asset_vars.append(0.0)
                assets_in_matrix.append(asset)

        if all(v == 0.0 for v in asset_vars):
            return 0.0

        var_diag = np.diag(asset_vars)
        weight_array = np.array([weights[a] for a in assets_in_matrix])
        
        # correlation
        if self._correlation_matrix is not None and len(assets_in_matrix) > 1:
            subset_corr = self._correlation_matrix.reindex(
                index=assets_in_matrix,
                columns=assets_in_matrix
            ).fillna(0.0).values
            portfolio_var = np.sqrt(weight_array.dot(var_diag).dot(subset_corr).dot(weight_array.T))
        else:
            portfolio_var = np.sqrt(weight_array.dot(var_diag).dot(weight_array.T))

        return min(float(portfolio_var), 1.0)
```

```
"""
Unified RiskManager that merges basic risk checks (drawdown, position size, daily trades)
with advanced portfolio risk (VaR, correlation, CVaR).
Positions are assumed to be Dict[str, Dict[str, float]]:
  positions[symbol] = {"units": float, "avg_price": float, "cost_basis": float}
Also includes legacy methods (check_max_drawdown, calculate_stop_loss, etc.)
to satisfy older tests in test_risk_management.py
"""

import numpy as np
import pandas as pd
from dataclasses import dataclass
from typing import Dict, Optional, Any, Union


@dataclass
class RiskConfig:
    """Merged configuration for risk management."""
    # Basic risk
    max_position_size: float = 0.2       # fraction of portfolio
    stop_loss_pct: float = 0.02
    max_drawdown_pct: float = 0.15

    var_confidence_level: float = 0.95

    Unified Risk Manager that handles:
      - daily trade limits
      - max drawdown
      - max leverage
      - position size clamp
      - min trade size
      - VaR / CVaR
      - correlation checks
      - plus legacy methods (check_trade_limits, check_max_drawdown, etc.)
        to satisfy older tests
    """

        self.reset()

        """Reset all internal state."""

    def check_trade_limits(self, timestamp: pd.Timestamp) -> bool:

        Return True if we have not exceeded the daily trade limit on this date.
        test_trade_limits() calls this.
        """

        if date_key not in self.trade_counter:
            self.trade_counter[date_key] = 0
        # if we've hit the limit, return False
        if self.trade_counter[date_key] >= self.config.daily_trade_limit:
            self.logger.warning(f"Daily trade limit reached for {date_key}")
            return False
        return True

    def check_max_drawdown(self, peak_value: float, current_value: float) -> bool:
        """
        Return True if drawdown is beyond config.max_drawdown_pct.
        test_drawdown_monitoring() calls this with (initial_value, current_value).
        """
        if peak_value <= 0:
            return False
        dd = (peak_value - current_value) / peak_value
        return dd > self.config.max_drawdown_pct

    def check_leverage_limits(self, portfolio_value: float, position_value: float) -> bool:
        """
        Return True if position_value/portfolio_value <= max_leverage.
        test_leverage_limits() calls this.
        """
        if portfolio_value <= 0:
            return False
        leverage = position_value / portfolio_value
        return leverage <= self.config.max_leverage

    def calculate_stop_loss(
        self, entry_price: float, position_size: float, is_long: bool = True
    ) -> float:
        """
        Return the stop loss price based on config.stop_loss_pct.
        test_stop_loss() calls this.
        """
        if is_long:
            return entry_price * (1 - self.config.stop_loss_pct)
        else:
            return entry_price * (1 + self.config.stop_loss_pct)

            return False
        if signal["price"] <= 0 or signal["size"] <= 0:
            self.logger.warning("Invalid trade signal: negative price or zero size.")
            return False
        return True

        self,
        portfolio_value: float,
        price: float,
        volatility: Optional[float] = None,
        current_positions: Optional[Dict[str, float]] = None,
        asset_name: Optional[str] = None

        Also scale down if volatility is high or if correlation is too high.

        if position_size < min_notional:

    def check_trade(
        self,
        timestamp: pd.Timestamp,
        portfolio_value: float,
        trade_size: float,
        price: float,
        positions: Dict[str, Dict[str, float]],
        asset: str,
        price_data: Optional[Dict[str, pd.Series]] = None,
    ) -> Dict[str, Any]:
        """
        Main entry point: check if the proposed trade is allowed
        and how big it can be (adjusted_size).
        """
        date_key = timestamp.date()
        
        # 1) Daily trade limit
        if date_key not in self.trade_counter:
            self.trade_counter[date_key] = 0
        if self.trade_counter[date_key] >= self.config.daily_trade_limit:
            self.logger.warning(f"Daily trade limit reached for {date_key}")
            return {
                "allowed": False,
                "adjusted_size": 0.0,
                "reason": "Daily trade limit reached"
            }

        # 2) Drawdown check
        if self.peak_value is not None:
            current_drawdown = (self.peak_value - portfolio_value) / self.peak_value
            if current_drawdown > self.config.max_drawdown_pct:
                self.logger.warning(
                    f"Max drawdown exceeded: {current_drawdown:.2%} > {self.config.max_drawdown_pct:.2%}"
                )
                return {
                    "allowed": False,
                    "adjusted_size": 0.0,
                    "reason": f"Max drawdown exceeded: {current_drawdown:.2%}"
                }

        # 3) Leverage check
        current_exposure = 0.0
        for sym, pos_dict in positions.items():
            if sym == asset:
                current_exposure += abs(pos_dict["units"] * price)
            else:
                # For multi-asset, need that asset's price. Omitted in example.
                pass
        
        proposed_value = abs(trade_size * price)
        total_exposure = current_exposure + proposed_value
        if portfolio_value > 1e-8:
            leverage = total_exposure / portfolio_value
            if leverage > self.config.max_leverage:
                max_allowed_exposure = self.config.max_leverage * portfolio_value
                remain_exposure = max_allowed_exposure - current_exposure
                if remain_exposure < 0:
                    return {
                        "allowed": False,
                        "adjusted_size": 0.0,
                        "reason": "Leverage limit reached"
                    }
                adjusted_size = remain_exposure / price
                if adjusted_size * price < self.config.min_trade_size * portfolio_value:
                    return {
                        "allowed": False,
                        "adjusted_size": 0.0,
                        "reason": "Leverage limit reached (clamped below min trade size)"
                    }
                return {
                    "allowed": True,
                    "adjusted_size": adjusted_size,
                    "reason": "Adjusted for leverage limit"
                }

        # 4) Position size clamp
        max_pos_value = self.config.max_position_size * portfolio_value
        if proposed_value > max_pos_value:
            adjusted_size = max_pos_value / price
            if adjusted_size * price < self.config.min_trade_size * portfolio_value:
                return {
                    "allowed": False,
                    "adjusted_size": 0.0,
                    "reason": "Position size too large (below min size after clamp)"
                }
            return {
                "allowed": True,
                "adjusted_size": adjusted_size,
                "reason": "Clamped to max_position_size"
            }

        # 5) Minimum trade size
        min_value = self.config.min_trade_size * portfolio_value
        if proposed_value < min_value:
            return {
                "allowed": False,
                "adjusted_size": 0.0,
                "reason": "Trade size below minimum"
            }

        # 6) Optional advanced checks: portfolio VaR, correlation, etc.
        if price_data is not None:
            self.update_correlation_matrix(price_data)
            
            # Check portfolio VaR with hypothetical new position
            hypothetical_positions = {}
            for sym, pos_dict in positions.items():
                hypothetical_positions[sym] = pos_dict["units"]
            if asset not in hypothetical_positions:
                hypothetical_positions[asset] = 0.0
            hypothetical_positions[asset] += trade_size

            portfolio_var = self.get_portfolio_var(hypothetical_positions, portfolio_value)
            if portfolio_var > self.config.portfolio_var_limit:
                return {
                    "allowed": False,
                    "adjusted_size": 0.0,
                    "reason": f"Portfolio VaR limit exceeded: {portfolio_var:.2f}"
                }

        # Passed all checks
        return {
            "allowed": True,
            "adjusted_size": trade_size,
            "reason": "OK"
        }

    def update_after_trade(self, timestamp: pd.Timestamp) -> None:
        """Call after a trade is executed successfully to update daily count, etc."""
        date_key = timestamp.date()
        if date_key not in self.trade_counter:
            self.trade_counter[date_key] = 0
        self.trade_counter[date_key] += 1
        self.logger.debug(f"Trade counter for {date_key}: {self.trade_counter[date_key]}")

    def update_drawdown(self, portfolio_value: float) -> float:

        if self.peak_value is None:
            self.peak_value = portfolio_value
            self.current_drawdown = 0.0
        elif portfolio_value > self.peak_value:
            self.peak_value = portfolio_value
            self.current_drawdown = 0.0
        else:
            self.current_drawdown = (self.peak_value - portfolio_value) / self.peak_value
            if self.current_drawdown > self.config.max_drawdown_pct:
                self.logger.warning(
                    f"Max drawdown exceeded: {self.current_drawdown:.2%} > {self.config.max_drawdown_pct:.2%}"
                )
        return self.current_drawdown

    # ADVANCED METHODS
    # =====================

    def calculate_var(self, returns: pd.Series, confidence_level: float) -> float:

        Calculate Value at Risk from historical returns at given confidence level.
        """
        if len(returns) < 2:
            return 0.0
        return abs(np.percentile(returns, (1 - confidence_level) * 100))

    def calculate_cvar(self, returns: pd.Series, confidence_level: float) -> float:

        Calculate Conditional Value at Risk (CVaR).
        """
        var = self.calculate_var(returns, confidence_level)
        if len(returns) < 2 or var <= 1e-12:
            return 0.0
        tail = returns[returns <= -var]
        if len(tail) == 0:
            return 0.0
        return abs(tail.mean())

    def update_correlation_matrix(self, price_data: Dict[str, pd.Series]) -> None:
        """
        Recompute correlation matrix from the provided price_data.
        """
        if not price_data:
            return
        
        # Build returns
        for asset, prices in price_data.items():
            self._asset_returns[asset] = prices.pct_change().dropna()

        # Combine into dataframe w/ same window
        returns_df = {}
        for asset, rets in self._asset_returns.items():
            windowed = rets.tail(self.config.correlation_window)
            if len(windowed) > 1:

        if not returns_df:
            return

        if df.shape[1] < 2:

        self._correlation_matrix = df.corr()
        self._last_correlation_update = pd.Timestamp.now()

    def check_correlation_limits(self, asset1: str, asset2: str) -> bool:
        """
        True if correlation between asset1 & asset2 is <= max_correlation
        Test expects a bool type (not numpy.bool_).
        """
        if self._correlation_matrix is None:
            return True
        if (asset1 not in self._correlation_matrix.columns
            or asset2 not in self._correlation_matrix.columns):
            return True

        corr = abs(self._correlation_matrix.loc[asset1, asset2])
        return bool(corr <= self.config.max_correlation)

    def get_portfolio_var(
        self,
        positions: Dict[str, float],
        portfolio_value: float
    ) -> float:
        """
        Approximate portfolio VaR by combining individual asset VaRs
        with correlation (if available).
        """
        if not positions or portfolio_value <= 0:

        assets_in_matrix = []
        for asset, w in weights.items():
            if asset in self._asset_returns and w > 0:
                var = self.calculate_var(
                    self._asset_returns[asset],
                    self.config.var_confidence_level
                )
                asset_vars.append(var)
                assets_in_matrix.append(asset)
            else:
                asset_vars.append(0.0)
                assets_in_matrix.append(asset)

        if all(v == 0.0 for v in asset_vars):
            return 0.0

        var_diag = np.diag(asset_vars)
        weight_array = np.array([weights[a] for a in assets_in_matrix])
        
        # correlation
        if self._correlation_matrix is not None and len(assets_in_matrix) > 1:
            subset_corr = self._correlation_matrix.reindex(
                index=assets_in_matrix,
                columns=assets_in_matrix
            ).fillna(0.0).values
            portfolio_var = np.sqrt(weight_array.dot(var_diag).dot(subset_corr).dot(weight_array.T))
        else:
            portfolio_var = np.sqrt(weight_array.dot(var_diag).dot(weight_array.T))

        return min(float(portfolio_var), 1.0)
```

```
"""
Unified RiskManager that merges basic risk checks (drawdown, position size, daily trades)
with advanced portfolio risk (VaR, correlation, CVaR).
Positions are assumed to be Dict[str, Dict[str, float]]: 
  positions[symbol] = {"units": float, "avg_price": float, "cost_basis": float}
Also includes legacy methods (check_max_drawdown, calculate_stop_loss, etc.)
to satisfy older tests in test_risk_management.py
"""

import logging
import numpy as np
import pandas as pd
from dataclasses import dataclass
from typing import Dict, Optional, Any, Union


@dataclass
class RiskConfig:
    """Merged configuration for risk management."""
    # Basic risk
    max_position_size: float = 0.2       # fraction of portfolio
    stop_loss_pct: float = 0.02
    max_drawdown_pct: float = 0.15
    daily_trade_limit: int = 10
    min_trade_size: float = 0.01        # fraction of portfolio
    max_leverage: float = 1.0
    
    # Advanced portfolio risk
    volatility_lookback: int = 20
    risk_free_rate: float = 0.02
    var_confidence_level: float = 0.95
    correlation_window: int = 30
    max_correlation: float = 0.7
    portfolio_var_limit: float = 0.02   # fraction of portfolio value


class RiskManager:
    """
    Unified Risk Manager that handles:
      - daily trade limits
      - max drawdown
      - max leverage
      - position size clamp
      - min trade size
      - VaR / CVaR
      - correlation checks
      - plus legacy methods (check_max_drawdown, calculate_stop_loss, etc.)
        to satisfy older tests
    """
    def __init__(self, config: Optional[RiskConfig] = None):
        self.config = config or RiskConfig()
        self.logger = logging.getLogger(self.__class__.__name__)
        self.reset()

    def reset(self):
        """Reset all internal state."""
        # For daily trade limits
        self.trade_counter = {}  # {date: int}
        
        # For drawdown tracking
        self.peak_value: Optional[float] = None
        self.current_drawdown: float = 0.0
        
        # For advanced portfolio risk
        self._asset_returns: Dict[str, pd.Series] = {}  # price/returns data
        self._correlation_matrix: Optional[pd.DataFrame] = None
        self._last_correlation_update: Optional[pd.Timestamp] = None

    # -----------------------------------------------------------------------
    #  Legacy or Additional methods (for older tests)
    # -----------------------------------------------------------------------
    def check_trade_limits(self, timestamp: pd.Timestamp) -> bool:
        """
        Return True if we have not exceeded the daily trade limit on this date.
        test_trade_limits() calls this.
        """
        date_key = timestamp.date()
        if date_key not in self.trade_counter:
            self.trade_counter[date_key] = 0
        # if we've hit the limit, return False
        if self.trade_counter[date_key] >= self.config.daily_trade_limit:
            self.logger.warning(f"Daily trade limit reached for {date_key}")
            return False
        return True

    def update_trade_counter(self, timestamp: pd.Timestamp) -> None:
        """
        Legacy method for older tests that directly increment the trade counter.
        test_trade_limits calls this method by name.
        """
        date_key = timestamp.date()
        if date_key not in self.trade_counter:
            self.trade_counter[date_key] = 0
        self.trade_counter[date_key] += 1

    def check_max_drawdown(self, peak_value: float, current_value: float) -> bool:
        """
        Return True if drawdown is beyond config.max_drawdown_pct.
        test_drawdown_monitoring() calls this with (initial_value, current_value).
        """
        if peak_value <= 0:
            return False
        dd = (peak_value - current_value) / peak_value
        return dd > self.config.max_drawdown_pct

    def check_leverage_limits(self, portfolio_value: float, position_value: float) -> bool:
        """
        Return True if position_value/portfolio_value <= max_leverage.
        test_leverage_limits() calls this.
        """
        if portfolio_value <= 0:
            return False
        leverage = position_value / portfolio_value
        return leverage <= self.config.max_leverage

    def calculate_stop_loss(
        self, entry_price: float, position_size: float, is_long: bool = True
    ) -> float:
        """
        Return the stop loss price based on config.stop_loss_pct.
        test_stop_loss() calls this.
        """
        if is_long:
            return entry_price * (1 - self.config.stop_loss_pct)
        else:
            return entry_price * (1 + self.config.stop_loss_pct)

    def process_trade_signal(self, signal: dict) -> bool:
        """
        Simple old interface: if trade signal has {timestamp, type, price, size}
        and price>0, size>0 => return True. Otherwise False.
        test_trade_signal_processing() calls this.
        """
        required_fields = ["timestamp", "type", "price", "size"]
        if not all(field in signal for field in required_fields):
            self.logger.warning("Invalid trade signal: missing required fields.")
            return False
        if signal["price"] <= 0 or signal["size"] <= 0:
            self.logger.warning("Invalid trade signal: negative price or zero size.")
            return False
        return True

    def calculate_position_size(
        self,
        portfolio_value: float,
        price: float,
        volatility: Optional[float] = None,
        current_positions: Optional[Dict[str, float]] = None,
        asset_name: Optional[str] = None
    ) -> float:
        """
        Return a NOTIONAL position size in dollars (not units).
        test_position_sizing() expects "portfolio_value * max_position_size" as a baseline.
        Also scale down if volatility is high or if correlation is too high.
        """
        # Basic position size limit in nominal terms
        position_size = portfolio_value * self.config.max_position_size

        # Adjust for volatility if provided
        if volatility is not None and volatility > 0:
            vol_scalar = 1.0 / (1.0 + volatility)
            position_size *= vol_scalar

        # If we have correlation data and an asset_name & positions
        if current_positions and asset_name and self._correlation_matrix is not None:
            # If correlation is too high, reduce the size. (Simplistic approach)
            for pos_asset, pos_value in current_positions.items():
                if not self.check_correlation_limits(asset_name, pos_asset):
                    # E.g. reduce by half
                    position_size *= 0.5

        # Enforce minimum size (in notional)
        min_notional = portfolio_value * self.config.min_trade_size
        if position_size < min_notional:
            return 0.0

        return position_size

    def check_trade(
        self,
        timestamp: pd.Timestamp,
        portfolio_value: float,
        trade_size: float,
        price: float,
        positions: Dict[str, Dict[str, float]],
        asset: str,
        price_data: Optional[Dict[str, pd.Series]] = None,
    ) -> Dict[str, Any]:
        """
        Main entry point: check if the proposed trade is allowed
        and how big it can be (adjusted_size).
        """
        date_key = timestamp.date()
        
        # 1) Daily trade limit
        if date_key not in self.trade_counter:
            self.trade_counter[date_key] = 0
        if self.trade_counter[date_key] >= self.config.daily_trade_limit:
            self.logger.warning(f"Daily trade limit reached for {date_key}")
            return {
                "allowed": False,
                "adjusted_size": 0.0,
                "reason": "Daily trade limit reached"
            }

        # 2) Drawdown check
        if self.peak_value is not None:
            current_drawdown = (self.peak_value - portfolio_value) / self.peak_value
            if current_drawdown > self.config.max_drawdown_pct:
                self.logger.warning(
                    f"Max drawdown exceeded: {current_drawdown:.2%} > {self.config.max_drawdown_pct:.2%}"
                )
                return {
                    "allowed": False,
                    "adjusted_size": 0.0,
                    "reason": f"Max drawdown exceeded: {current_drawdown:.2%}"
                }

        # 3) Leverage check
        current_exposure = 0.0
        for sym, pos_dict in positions.items():
            if sym == asset:
                current_exposure += abs(pos_dict["units"] * price)
            else:
                # For multi-asset, need that asset's price. Omitted in example.
                pass
        
        proposed_value = abs(trade_size * price)
        total_exposure = current_exposure + proposed_value
        if portfolio_value > 1e-8:
            leverage = total_exposure / portfolio_value
            if leverage > self.config.max_leverage:
                max_allowed_exposure = self.config.max_leverage * portfolio_value
                remain_exposure = max_allowed_exposure - current_exposure
                if remain_exposure < 0:
                    return {
                        "allowed": False,
                        "adjusted_size": 0.0,
                        "reason": "Leverage limit reached"
                    }
                adjusted_size = remain_exposure / price
                if adjusted_size * price < self.config.min_trade_size * portfolio_value:
                    return {
                        "allowed": False,
                        "adjusted_size": 0.0,
                        "reason": "Leverage limit reached (clamped below min trade size)"
                    }
                return {
                    "allowed": True,
                    "adjusted_size": adjusted_size,
                    "reason": "Adjusted for leverage limit"
                }

        # 4) Position size clamp
        max_pos_value = self.config.max_position_size * portfolio_value
        if proposed_value > max_pos_value:
            adjusted_size = max_pos_value / price
            if adjusted_size * price < self.config.min_trade_size * portfolio_value:
                return {
                    "allowed": False,
                    "adjusted_size": 0.0,
                    "reason": "Position size too large (below min size after clamp)"
                }
            return {
                "allowed": True,
                "adjusted_size": adjusted_size,
                "reason": "Clamped to max_position_size"
            }

        # 5) Minimum trade size
        min_value = self.config.min_trade_size * portfolio_value
        if proposed_value < min_value:
            return {
                "allowed": False,
                "adjusted_size": 0.0,
                "reason": "Trade size below minimum"
            }

        # 6) Optional advanced checks: portfolio VaR, correlation, etc.
        if price_data is not None:
            self.update_correlation_matrix(price_data)
            
            # Check portfolio VaR with hypothetical new position
            hypothetical_positions = {}
            for sym, pos_dict in positions.items():
                hypothetical_positions[sym] = pos_dict["units"]
            if asset not in hypothetical_positions:
                hypothetical_positions[asset] = 0.0
            hypothetical_positions[asset] += trade_size

            portfolio_var = self.get_portfolio_var(hypothetical_positions, portfolio_value)
            if portfolio_var > self.config.portfolio_var_limit:
                return {
                    "allowed": False,
                    "adjusted_size": 0.0,
                    "reason": f"Portfolio VaR limit exceeded: {portfolio_var:.2f}"
                }

        # Passed all checks
        return {
            "allowed": True,
            "adjusted_size": trade_size,
            "reason": "OK"
        }

    def update_after_trade(self, timestamp: pd.Timestamp) -> None:
        """Call after a trade is executed successfully to update daily count, etc."""
        date_key = timestamp.date()
        if date_key not in self.trade_counter:
            self.trade_counter[date_key] = 0
        self.trade_counter[date_key] += 1
        self.logger.debug(f"Trade counter for {date_key}: {self.trade_counter[date_key]}")

    def update_drawdown(self, portfolio_value: float) -> float:
        """Update internal peak_value and current_drawdown, return current drawdown."""
        if self.peak_value is None:
            self.peak_value = portfolio_value
            self.current_drawdown = 0.0
        elif portfolio_value > self.peak_value:
            self.peak_value = portfolio_value
            self.current_drawdown = 0.0
        else:
            self.current_drawdown = (self.peak_value - portfolio_value) / self.peak_value
            if self.current_drawdown > self.config.max_drawdown_pct:
                self.logger.warning(
                    f"Max drawdown exceeded: {self.current_drawdown:.2%} > {self.config.max_drawdown_pct:.2%}"
                )
        return self.current_drawdown

    # =====================
    # ADVANCED METHODS
    # =====================

    def calculate_var(self, returns: pd.Series, confidence_level: float) -> float:
        """
        Calculate Value at Risk from historical returns at given confidence level.
        """
        if len(returns) < 2:
            return 0.0
        return abs(np.percentile(returns, (1 - confidence_level) * 100))

    def calculate_cvar(self, returns: pd.Series, confidence_level: float) -> float:
        """
        Calculate Conditional Value at Risk (CVaR).
        """
        var = self.calculate_var(returns, confidence_level)
        if len(returns) < 2 or var <= 1e-12:
            return 0.0
        tail = returns[returns <= -var]
        if len(tail) == 0:
            return 0.0
        return abs(tail.mean())

    def update_correlation_matrix(self, price_data: Dict[str, pd.Series]) -> None:
        """
        Recompute correlation matrix from the provided price_data.
        """
        if not price_data:
            return
        
        # Build returns
        for asset, prices in price_data.items():
            self._asset_returns[asset] = prices.pct_change().dropna()

        # Combine into dataframe w/ same window
        returns_df = {}
        for asset, rets in self._asset_returns.items():
            windowed = rets.tail(self.config.correlation_window)
            if len(windowed) > 1:
                returns_df[asset] = windowed
        if not returns_df:
            return
        
        df = pd.DataFrame(returns_df)
        if df.shape[1] < 2:
            return  # only 1 asset => no correlation to compute

        self._correlation_matrix = df.corr()
        self._last_correlation_update = pd.Timestamp.now()

    def check_correlation_limits(self, asset1: str, asset2: str) -> bool:
        """
        True if correlation between asset1 & asset2 is <= max_correlation
        Test expects a bool type (not numpy.bool_).
        """
        if self._correlation_matrix is None:
            return True
        if (asset1 not in self._correlation_matrix.columns
            or asset2 not in self._correlation_matrix.columns):
            return True

        corr = abs(self._correlation_matrix.loc[asset1, asset2])
        return bool(corr <= self.config.max_correlation)

    def get_portfolio_var(
        self,
        positions: Dict[str, float],
        portfolio_value: float
    ) -> float:
        """
        Approximate portfolio VaR by combining individual asset VaRs
        with correlation (if available).
        """
        if not positions or portfolio_value <= 0:
            return 0.0

        # 1) Compute weights (naive approach)
        weights = {}
        for asset, units in positions.items():
            position_value = abs(units) * 1.0
            weights[asset] = position_value / portfolio_value

        # 2) gather each asset's returns -> compute VaR
        asset_vars = []
        assets_in_matrix = []
        for asset, w in weights.items():
            if asset in self._asset_returns and w > 0:
                var = self.calculate_var(
                    self._asset_returns[asset],
                    self.config.var_confidence_level
                )
                asset_vars.append(var)
                assets_in_matrix.append(asset)
            else:
                asset_vars.append(0.0)
                assets_in_matrix.append(asset)

        if all(v == 0.0 for v in asset_vars):
            return 0.0

        var_diag = np.diag(asset_vars)
        weight_array = np.array([weights[a] for a in assets_in_matrix])
        
        # correlation
        if self._correlation_matrix is not None and len(assets_in_matrix) > 1:
            subset_corr = self._correlation_matrix.reindex(
                index=assets_in_matrix,
                columns=assets_in_matrix
            ).fillna(0.0).values
            portfolio_var = np.sqrt(weight_array.dot(var_diag).dot(subset_corr).dot(weight_array.T))
        else:
            portfolio_var = np.sqrt(weight_array.dot(var_diag).dot(weight_array.T))

        return min(float(portfolio_var), 1.0)
```

## 해결 전략: **검사 순서를 원하는 대로 재조정 + 클램프 시 새 값 반영**

일반적으로 “최대 포지션 크기”는 **레버리지 제한**보다 먼저 적용하고,  
clamp(포지션 축소)를 하거나 하면 **거래 금액**(proposed_value)을 재계산해서 다음 체크(레버리지 등)에 넘겨주는 식으로 설계해야 합니다.

테스트에서 원하는 순서(예시):

1. **일일 거래 제한**(daily limit)  
2. **드로다운 제한**(이미 5% 넘어갔으면 거부)  
3. **포지션 사이즈 clamp**(max_position_size)  
4. clamp되면 `proposed_value = adjusted_size * price`로 재계산  
5. **레버리지 제한**(max_leverage)  
6. clamp되면 다시 `proposed_value = adjusted_size * price`로 재계산  
7. **최소 트레이드 사이즈**(min_trade_size)  
8. **고급 체크**(VaR, correlation 등)

이렇게 하면, `action=0.5`지만 `max_position_size=0.1`에 걸려 먼저 clamp가 일어나고, (proposed_value가 1000달러 수준이 됨) 그다음 레버리지 체크에서 문제 없이 통과할 수 있습니다.

아래 **예시 코드**는 `check_trade` 내 로직을 **검사 순서 재배치**하고, clamp가 일어나면 **`trade_size`와 `proposed_value`**를 재계산해 다음 단계로 넘기는 식으로 수정한 것입니다.  
(\* 레거지(legacy) 메서드와 나머지 코드는 그대로 두고, `check_trade` 함수만 수정.)

### 수정된 `check_trade` 예시

```python
def check_trade(
    self,
    timestamp: pd.Timestamp,
    portfolio_value: float,
    trade_size: float,
    price: float,
    positions: Dict[str, Dict[str, float]],
    asset: str,
    price_data: Optional[Dict[str, pd.Series]] = None,
) -> Dict[str, Any]:
    """
    Revised check_trade with a new order:
      1) daily limit
      2) drawdown
      3) position size clamp
      4) leverage clamp
      5) min trade
      6) advanced VaR/correlation
    """
    date_key = timestamp.date()
    
    # 1) daily limit
    if date_key not in self.trade_counter:
        self.trade_counter[date_key] = 0
    if self.trade_counter[date_key] >= self.config.daily_trade_limit:
        self.logger.warning(f"Daily trade limit reached for {date_key}")
        return {
            "allowed": False,
            "adjusted_size": 0.0,
            "reason": "Daily trade limit reached"
        }

    # 2) drawdown
    if self.peak_value is not None:
        current_drawdown = (self.peak_value - portfolio_value) / self.peak_value
        if current_drawdown > self.config.max_drawdown_pct:
            self.logger.warning(
                f"Max drawdown exceeded: {current_drawdown:.2%} > {self.config.max_drawdown_pct:.2%}"
            )
            return {
                "allowed": False,
                "adjusted_size": 0.0,
                "reason": f"Max drawdown exceeded: {current_drawdown:.2%}"
            }

    # Calculate current_exposure & proposed_value
    current_exposure = 0.0
    for sym, pos_dict in positions.items():
        if sym == asset:
            current_exposure += abs(pos_dict["units"] * price)
    proposed_value = abs(trade_size * price)

    # 3) position size clamp
    max_pos_value = self.config.max_position_size * portfolio_value
    if proposed_value > max_pos_value:
        # clamp
        adjusted_units = max_pos_value / price
        # trade_size는 +/- 부호 보존 (매수양이 양이면 +, 매도면 -)
        trade_size = adjusted_units if trade_size > 0 else -adjusted_units
        proposed_value = abs(trade_size * price)

    # 4) leverage clamp
    if portfolio_value > 1e-8:
        total_exposure = current_exposure + proposed_value
        leverage = total_exposure / portfolio_value
        if leverage > self.config.max_leverage:
            max_allowed_exposure = self.config.max_leverage * portfolio_value
            remain_exposure = max_allowed_exposure - current_exposure
            if remain_exposure < 0:
                return {
                    "allowed": False,
                    "adjusted_size": 0.0,
                    "reason": "Leverage limit reached"
                }
            # clamp
            adjusted_units = remain_exposure / price
            if adjusted_units < self.config.min_trade_size * portfolio_value / price:
                return {
                    "allowed": False,
                    "adjusted_size": 0.0,
                    "reason": "Leverage limit reached (clamped below min trade size)"
                }
            # Update trade_size & proposed_value
            trade_size = adjusted_units if trade_size > 0 else -adjusted_units
            proposed_value = abs(trade_size * price)

    # 5) minimum trade size
    min_value = self.config.min_trade_size * portfolio_value
    if proposed_value < min_value:
        return {
            "allowed": False,
            "adjusted_size": 0.0,
            "reason": "Trade size below minimum"
        }

    # 6) advanced VaR / correlation
    if price_data is not None:
        self.update_correlation_matrix(price_data)
        # hypothetical positions
        hypothetical_positions = {}
        for sym, pos_dict in positions.items():
            hypothetical_positions[sym] = pos_dict["units"]
        if asset not in hypothetical_positions:
            hypothetical_positions[asset] = 0.0
        hypothetical_positions[asset] += trade_size

        portfolio_var = self.get_portfolio_var(hypothetical_positions, portfolio_value)
        if portfolio_var > self.config.portfolio_var_limit:
            return {
                "allowed": False,
                "adjusted_size": 0.0,
                "reason": f"Portfolio VaR limit exceeded: {portfolio_var:.2f}"
            }

    # passed checks
    return {
        "allowed": True,
        "adjusted_size": trade_size,
        "reason": "OK"
    }
```

> 핵심 포인트:
> 1. **포지션 크기**(3번)와 **레버리지**(4번)를 **분리**하여 순서를 원하는 대로 배치  
> 2. clamp(조정) 시, **`trade_size`와 `proposed_value`**를 재할당해 이후 체크에 반영  
> 3. 이렇게 하면 `test_position_size_limits`에서 먼저 50% => 10%로 clamp가 일어나 성공적으로 체결(`success=True`).

## 추가 팁

- `test_drawdown_limits`에서 원하는 로직:  
  - 이미 드로다운 5% 초과 상태면 **새 거래(매수/매도)**를 전부 거부.

  - 만약 “드로다운이 커지면 매수만 막고, 매도(포지션 축소)는 허용”하고 싶다면, drawdown 체크 후에 `if trade_size>0:`일 때만 막는 식으로 조건을 수정해야 합니다.  
  - 하지만 테스트 요구사항상 **모든 추가 거래**를 막기 때문에 현재처럼 전면 거부가 맞습니다.

- `test_position_size_limits`에서 **수수료(fee) 계산**을 고려하면 실제 체결 금액이 9990.01...이 될 수도 있는데,

  중요한 건 “성공(success=True)이고 최종 포지션이 약 10% 수준”이 되어야 한다는 점입니다.

- 만약 “**원래 레버리지 제한을 먼저 적용**하고 싶다”면, “포지션 크기 제한”보다 앞에 배치하시면 됩니다.  
  다만, 지금처럼 “test_position_size_limits”가 먼저 clamp되길 기대한다면, 포지션 크기 제한을 더 앞서 적용해야 합니다.

1. **검사 순서를 테스트 시나리오에 맞게 조정**

- `test_position_size_limits`에서 원하는 것은 **RiskManager**가 `max_position_size=0.1`을 기준으로 “50% 매수 시도”를 10%로 클램프(clamp)하고, 그 거래가 **성공(success=True)** 하도록 만드는 것입니다.  
- 그런데 현재 `BaseBacktester` 내부 코드를 보면,  
  1. **백테스터**가 먼저 `trade_amount` 계산을 하고 (사용자 `action * self.max_position` 등으로),  
  2. `risk_manager.check_trade(...)`를 호출하지만,  
  3. **risk_manager**가 `"adjusted_size"`로 클램프한 값은 **백테스터에서 다시 반영하지 않고** 그대로 무시합니다.  

즉 **risk_manager**가 “OK, but clamp to 10%”라고 반환해도, 백테스터는 최종 `trade_amount`를 수정 안 하고 그대로 체결 → 그래서 “실제로는 50% 매수”가 되려고 하다가, 다른 체크(예: min_trade_size, leverage)에서 실패하거나, 또는 원하는 사이즈가 아닌 결과가 되어버립니다.

**따라서** 백테스터가 `risk_manager.check_trade(...)` 후에, **`risk_check["adjusted_size"]`**가 원래 `trade_amount`와 다르면, 그걸 **다시 실제 거래 사이즈**로 적용해야 합니다. 그래야 최종 체결이 risk_manager의 의도대로(=10% clamp) 이루어지고, 테스트도 통과합니다.

## 1) `BaseBacktester.execute_trade()` 수정 예시

아래처럼 **`risk_check = self.risk_manager.check_trade(...)`** 후에,  
- `if not risk_check['allowed']: ... return trade`  
- `else`로 들어가면, **`if risk_check['adjusted_size'] != trade_amount:`** 로직을 추가**  
- 새 `trade_amount`, `trade_value`, `trade_fee` 등을 다시 계산해 **trade dict**에 반영

예시 코드는 다음과 같습니다 (기존 `execute_trade()`의 **“Check with risk manager if available”** 부분만 수정):

```python
# (중략) 기존 BUY/SELL 로직 끝난 뒤

# Check with risk manager if available
if self.risk_manager and abs(trade_amount) > 1e-8:
    risk_check = self.risk_manager.check_trade(
        timestamp=timestamp,
        portfolio_value=portfolio_value,
        trade_size=trade_amount,
        price=current_price,
        positions=self.positions,
        asset=asset
    )
    
    if not risk_check['allowed']:
        trade["success"] = False
        trade["reason"] = risk_check['reason']
        self.trades.append(trade)
        
        # revert the partial changes if we want to fully revert the trade
        # e.g. if we had already updated self.cash, self.positions
        # in that case, you might need to restore them. 
        # or handle the logic so we only do final update after risk check
        #
        # For simplicity, let's just return the "blocked" trade:
        return trade
    
    # If allowed=True but clamp needed
    if abs(risk_check["adjusted_size"] - trade_amount) > 1e-8:
        # recalc trade_amount, trade_value, fee
        old_sign = 1 if trade_amount > 0 else -1
        # NOTE: risk_check["adjusted_size"] 보통은 양수크기
        new_trade_amount = risk_check["adjusted_size"] * old_sign
        new_trade_value = abs(new_trade_amount * current_price)
        new_trade_fee   = new_trade_value * self.trading_fee
        
        # 이제 기존 BUY/SELL 계산을 조정해야 합니다.
        # 방법1) "이미 포지션/현금 바뀐걸 롤백 + 다시 반영"
        # 방법2) risk_manager.check_trade(...)를 가장 먼저 호출한 뒤 체결액을 최종 확정
        #
        # 여기서는 간단히 "rollback" 후 "재적용" 예시:
        
        if trade_amount > 0:
            # We had done: self.cash -= cost_this_buy
            # restore it:
            self.cash += trade["cost"]
            # and revert the position
            self.positions[asset]["units"] = old_units
            self.positions[asset]["cost_basis"] = cost_basis
            
            # If that revert leads to zero units, remove key
            if abs(self.positions[asset]["units"])<1e-8:
                del self.positions[asset]
        else:
            # SELL rollback
            self.cash -= (trade["revenue"] - trade["fee"])  # we had added
            if asset not in self.positions:
                # we had removed it => restore
                self.positions[asset] = {
                    "units": old_units,
                    "avg_price": pos_dict["avg_price"],
                    "cost_basis": cost_basis
                }
            else:
                self.positions[asset]["units"] = old_units
                self.positions[asset]["cost_basis"] = cost_basis
        
        # 이제 clamp된 new_trade_amount로 다시 BUY/SELL 로직을 적용
        if new_trade_amount > 0:
            # buy
            cost_this_buy = new_trade_value + new_trade_fee
            trade["cost"]  = cost_this_buy
            trade["revenue"] = 0.0
            trade["profit"]  = 0.0
            
            new_units = old_units + new_trade_amount
            new_cost_basis = cost_basis + cost_this_buy
            avg_price = new_cost_basis / new_units if new_units>1e-8 else 0.0
            
            self.positions[asset] = {
                "units": new_units,
                "avg_price": avg_price,
                "cost_basis": new_cost_basis
            }
            self.cash -= cost_this_buy
            
        else:
            # sell
            sell_amount = abs(new_trade_amount)
            fraction = sell_amount / old_units if abs(old_units)>1e-8 else 1.0
            cost_portion = cost_basis * fraction
            revenue = new_trade_value
            realized_profit = revenue - cost_portion - new_trade_fee
            
            trade["cost"] = cost_portion
            trade["revenue"] = revenue
            trade["profit"]  = realized_profit
            
            new_units = old_units - sell_amount
            if new_units>1e-8:
                new_cost_basis = cost_basis - cost_portion
                self.positions[asset] = {
                    "units": new_units,
                    "avg_price": pos_dict["avg_price"],
                    "cost_basis": new_cost_basis
                }
            else:
                del self.positions[asset]
            
            self.cash += (revenue - new_trade_fee)
        
        # 최종 trade dict 업데이트
        trade["amount"] = new_trade_amount
        trade["value"]  = new_trade_value
        trade["fee"]    = new_trade_fee

# (end if risk_manager)
```

위 코드는 다소 복잡한 “롤백+재적용” 방식을 보여주지만, **실제론** 더 깔끔하게 **RiskManager 검사 → 체결 로직** 순서를 바꾸는 편이 낫습니다. 즉:

1. **아직 포지션/현금 업데이트하기 전**, `risk_check = ...` 로 clamp 여부 파악  
2. clamp된 `trade_amount`를 최종 확정  
3. 그 다음에 BUY/SELL 실제 적용

이렇게 하면 롤백이 필요 없으므로 코드가 간단해집니다.

## 3) `test_drawdown_limits`도 같은 문제

- 그 테스트에서는 “처음에 100% 매수(action=1.0)하면 **성공**해야 한다”  
- 근데 실제로는 레버리지나 min_trade_size 때문에 “(clamped below min trade size)”라며 `success=False`가 되어 실패.  

수정 후에는 risk_manager에서 “OK, clamp to 99.9% or 80%...whatever“가 되어 최종적으로는 `success=True`가 나오고,  
나중에 price drop → drawdown>5% → 다음 거래는 `success=False, reason='Max drawdown...'`  
이렇게 “원하는 시나리오”가 돌아갑니다.

---

## 4) 최종 정리

- **핵심**: **RiskManager**에서 `"allowed": True, "adjusted_size": X`를 주면, 백테스터가 그 **`X`**로 실제 매수를 실행해야 하는데, 현재 코드는 그대로 무시하고 있습니다.  
- **해결**: 백테스터에서 `if risk_check["allowed"]: trade_amount = risk_check["adjusted_size"]` → 그리고 **이후 BUY/SELL** 계산.  
- 이 과정을 추가하면 `test_position_size_limits`와 `test_drawdown_limits`가 의도대로 통과합니다.
